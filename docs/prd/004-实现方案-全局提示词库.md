```kotlin
    var content: String,
```

```kotlin
// PromptStorage.kt
@State(
    name = "com.dmc.kilo.prompt-collection",
    storages = [Storage("kilo-prompt-collection.xml")]
)
@Service
class PromptStorage : PersistentStateComponent<PromptStorage.State> {

    data class State(var items: MutableList<PromptItem> = mutableListOf())

    private var state = State()

    override fun getState() = state
    override fun loadState(loaded: State) { state = loaded }

    val items: List<PromptItem> get() = state.items.toList()

    fun add(item: PromptItem) {
        state.items.add(item)
    }

    fun update(id: String, block: (PromptItem) -> Unit) {
        val item = state.items.find { it.id == id } ?: return
        block(item)
        item.updateTime = System.currentTimeMillis()
    }

    fun remove(id: String) {
        state.items.removeAll { it.id == id }
    }

    companion object {
        fun getInstance(): PromptStorage =
            ApplicationManager.getApplication().getService(PromptStorage::class.java)
    }
}
```

### 模块 2：业务管理（PromptManager）

#### 实现

```kotlin
// PromptManager.kt
object PromptManager {

    private val storage get() = PromptStorage.getInstance()

    fun getAll(): List<PromptItem> =
        storage.items.sortedWith(compareByDescending<PromptItem> { it.isTop }.thenByDescending { it.updateTime })

    fun search(query: String): List<PromptItem> {
        if (query.isBlank()) return getAll()
        val q = query.lowercase()
        return getAll().filter {
            it.name.lowercase().contains(q) ||
            it.content.lowercase().contains(q) ||
            it.tags.any { tag -> tag.lowercase().contains(q) }
        }
    }

    fun getByCategory(category: String): List<PromptItem> =
        getAll().filter { it.category == category }

    fun getCategories(): List<String> =
        storage.items.map { it.category }.distinct().sorted()

    fun add(name: String, content: String, category: String = "通用"): PromptItem {
        val item = PromptItem(name = name, content = content, category = category)
        storage.add(item)
        return item
    }

    fun update(id: String, name: String? = null, content: String? = null, category: String? = null) {
        storage.update(id) {
            name?.let { it_ref -> it.name = name }
            content?.let { it_ref -> it.content = content }
            category?.let { it_ref -> it.category = category }
        }
    }

    fun remove(id: String) = storage.remove(id)

    fun toggleTop(id: String) {
        storage.update(id) { it.isTop = !it.isTop }
    }

    fun exportToJson(): String {
        val items = storage.items
        return GsonBuilder().setPrettyPrinting().create().toJson(items)
    }

    fun importFromJson(json: String, onConflict: (PromptItem) -> ImportAction): Int {
        val imported = Gson().fromJson(json, Array<PromptItem>::class.java) ?: return 0
        var count = 0
        for (item in imported) {
            val existing = storage.items.find { it.name == item.name }
            if (existing != null) {
                when (onConflict(item)) {
                    ImportAction.OVERWRITE -> { storage.remove(existing.id); storage.add(item.copy(id = java.util.UUID.randomUUID().toString())) }
                    ImportAction.SKIP -> continue
                    ImportAction.RENAME -> storage.add(item.copy(id = java.util.UUID.randomUUID().toString(), name = "${item.name} (导入)"))
                }
            } else {
                storage.add(item.copy(id = java.util.UUID.randomUUID().toString()))
            }
            count++
        }
        return count
    }

    enum class ImportAction { OVERWRITE, SKIP, RENAME }
}
```

```kotlin
class PromptListPanel(
    private val project: Project,
    private val onSelect: (PromptItem?) -> Unit,
) : JPanel(BorderLayout()), WikiToolWindowFactory.Refreshable {

    private val listModel = DefaultListModel<PromptItem>()
    private val promptList = JBList(listModel)
    private val searchField = JBTextField()

    init {
        // 搜索框
        searchField.emptyText.text = "搜索提示词..."
        searchField.addKeyListener(object : KeyAdapter() {
            override fun keyReleased(e: KeyEvent) = refresh(searchField.text.trim())
        })
        add(searchField, BorderLayout.NORTH)

        // 列表
        promptList.cellRenderer = PromptListCellRenderer()
        promptList.addListSelectionListener {
            if (!it.valueIsAdjusting) onSelect(promptList.selectedValue)
        }
        add(JBScrollPane(promptList), BorderLayout.CENTER)

        refresh("")
    }

    override fun refresh(query: String) {
        ApplicationManager.getApplication().executeOnPooledThread {
            val items = if (query.isEmpty()) PromptManager.getAll() else PromptManager.search(query)
            ApplicationManager.getApplication().invokeLater {
                listModel.clear()
                items.forEach { listModel.addElement(it) }
            }
        }
    }
}
```

#### 实现流程

```
用户点击 [AI优化]
  ↓
组装优化 Prompt:
  "你是专业AI提示词优化专家，请优化以下提示词：
   原文：{promptItem.content}
   要求：1.逻辑分层清晰 2.增加约束 3.适配代码场景 4.不改变核心诉求

  ↓

  ↓

  ↓
用户查看 → 复制 → 回到面板粘贴覆盖 OR 手动确认
```

#### 代码

```kotlin
// OptimizePromptAction.kt
class OptimizePromptAction(
    private val project: Project,
    private val getItem: () -> PromptItem?,
) : AnAction("AI 优化", "调用 Kilo AI 优化当前提示词", AllIcons.Actions.InlayGlobe) {

    override fun actionPerformed(e: AnActionEvent) {
        val item = getItem() ?: return

        val manager = DmcSessionResolver.getSessionManager(project)
        if (manager == null) {
            notify("请先打开 Kilo 工具窗口")
            return
        }

        val optimizePrompt = buildString {
            appendLine("你是专业 AI 提示词优化专家。请优化以下提示词，不改变核心诉求：")
            appendLine()
            appendLine("---")
            appendLine(item.content)
            appendLine("---")
            appendLine()
            appendLine("优化要求：")
            appendLine("1. 逻辑分层清晰，消除歧义")
            appendLine("2. 增加约束条件（输出格式、返回要求）")
            appendLine("3. 适配代码开发场景")
            appendLine("4. 返回优化后的完整提示词")
        }

        try {
            manager.sendPrompt(optimizePrompt)
            notify("已发送到 Kilo，AI 正在优化，请在对话区查看结果")
        } catch (ex: Exception) {
            notify("优化失败：${ex.message}")
        }
    }

    private fun notify(message: String) {
        Notification("Kilo Code", message, NotificationType.INFORMATION)
            .notify(project)
    }
}
```

```kotlin
// SaveCurrentPromptAction.kt
class SaveCurrentPromptAction : AnAction("收藏提示词", "收藏当前输入框内容到提示词库", AllIcons.Actions.AddList) {

    override fun update(e: AnActionEvent) {
        // 仅在 Kilo 提示词编辑器中显示
        val editor = e.getData(CommonDataKeys.EDITOR)
        val isKiloPrompt = e.getData(CommonDataKeys.VIRTUAL_FILE)?.path?.contains("prompt") == true
        e.presentation.isEnabledAndVisible = editor != null && editor.document.text.isNotEmpty()
    }

    override fun actionPerformed(e: AnActionEvent) {
        val project = e.project ?: return
        val editor = e.getData(CommonDataKeys.EDITOR) ?: return
        val text = editor.document.text.trim()

        if (text.isEmpty()) return

        // 弹窗输入名称和分类
        val name = Messages.showInputDialog(project, "提示词名称：", "收藏提示词", Messages.getQuestionIcon())
            ?: return
        if (name.isBlank()) return

        PromptManager.add(name, text)
        Notification("Kilo Code", "已收藏「$name」", NotificationType.INFORMATION).notify(project)
    }
}
```

```kotlin
// ImportExportAction.kt — 注册为 ToolWindow 底部按钮
fun exportPrompts(project: Project) {
    val json = PromptManager.exportToJson()
    val fileChooser = JFileChooser()
    fileChooser.selectedFile = java.io.File("kilo-prompts.json")
    if (fileChooser.showSaveDialog(null) == JFileChooser.APPROVE_OPTION) {
        fileChooser.selectedFile.writeText(json, Charsets.UTF_8)
        Notification("Kilo Code", "已导出 ${PromptManager.getAll().size} 条提示词", NotificationType.INFORMATION).notify(project)
    }
}

fun importPrompts(project: Project) {
    val fileChooser = JFileChooser()
    if (fileChooser.showOpenDialog(null) == JFileChooser.APPROVE_OPTION) {
        val json = fileChooser.selectedFile.readText(Charsets.UTF_8)
        val count = PromptManager.importFromJson(json) { _ -> PromptManager.ImportAction.RENAME }
        Notification("Kilo Code", "已导入 $count 条提示词", NotificationType.INFORMATION).notify(project)
    }
}
```
