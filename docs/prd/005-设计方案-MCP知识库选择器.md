# 设计方案：MCP 知识库选择器（PromptPanel 工具栏）

> 版本：v1 草案
> 日期：2026-08-13
> 状态：待审核

---

## 一、需求概述

在 Kilo 聊天输入框工具栏新增一个 MCP 选择按钮，用户点击后弹出选择器：
- 列出所有已配置的 MCP 服务器及其工具
- 支持服务器级勾选（优先使用该服务器所有工具）
- 支持工具级勾选（优先使用指定工具）
- 选择后，CLI 插件通过 `system.transform` 注入引导指令，让 AI 优先调用选中的 MCP 工具

---

### 3.2 完整代码

```typescript
import type { Plugin } from "@kilocode/plugin"
import { writeFileSync, readFileSync, existsSync, mkdirSync } from "node:fs"
import { join } from "node:path"

// 内置工具名（用于过滤 MCP 工具）
const BUILTIN_TOOLS = new Set([
  "read", "write", "edit", "bash", "grep", "glob",
  "webfetch", "websearch", "question", "task",
  "todowrite", "todoread", "plan", "skill",
  "agent_manager", "notify_user",
])

function ensureDir(dir: string) {
  if (!existsSync(dir)) mkdirSync(dir, { recursive: true })
}

function writeJson(path: string, data: any) {
  ensureDir(join(path, ".."))
  writeFileSync(path, JSON.stringify(data, null, 2), "utf-8")
}

function readJson(path: string): any | null {
  try {
    return JSON.parse(readFileSync(path, "utf-8"))
  } catch {
    return null
  }
}

const McpSelectorPlugin: Plugin = async ({ directory }) => {
  const cacheDir = join(directory, ".kilo", ".cache")
  const toolsPath = join(cacheDir, "mcp-tools.json")
  const activePath = join(cacheDir, "active-mcp.json")

  return {
    // ---- 捕获所有工具定义（含 MCP），写入缓存 ----
    "tool.definition": async (tools) => {
      try {
        // 过滤出 MCP 工具（非内置工具）
        const mcpTools = tools.filter(t => !BUILTIN_TOOLS.has(t.name))

        // 按 server 名分组（MCP 工具命名：{server}_{tool}）
        const serverMap = new Map<string, any[]>()
        for (const tool of mcpTools) {
          // 尝试解析 server 名：第一个下划线之前的部分
          const idx = tool.name.indexOf("_")
          const serverName = idx > 0 ? tool.name.substring(0, idx) : tool.name
          if (!serverMap.has(serverName)) {
            serverMap.set(serverName, [])
          }
          serverMap.get(serverName)!.push({
            name: tool.name,
            description: tool.description || "",
          })
        }

        const data = {
          updatedAt: new Date().toISOString(),
          servers: Array.from(serverMap.entries()).map(([name, tools]) => ({
            name,
            tools,
          })),
        }
        writeJson(toolsPath, data)
      } catch {
        // 静默失败，不影响对话
      }

      return tools // 原样返回，不修改工具定义
    },

    // ---- 每次对话前注入 MCP 选择引导 ----
    "experimental.chat.system.transform": async (_input, output) => {
      try {
        const active = readJson(activePath)
        if (active && active.selectedTools && active.selectedTools.length > 0) {
          const toolList = active.selectedTools.join(", ")
          const instruction = active.instruction || "请优先使用以下工具"
          output.system.push(
            `## MCP 工具优先使用规则\n${instruction}：${toolList}`
          )
        }
      } catch {
        // 静默失败
      }
    },
  }
}

export default { id: "mcp-selector", server: McpSelectorPlugin }
```

### 4.3 McpToolCache.kt

```kotlin
package com.dmc.prompt

import com.google.gson.Gson
import com.google.gson.reflect.TypeToken
import com.intellij.openapi.project.Project
import java.io.File

object McpToolCache {

    private val gson = Gson()

    data class McpTool(val name: String, val description: String)
    data class McpServer(val name: String, val tools: List<McpTool>)
    data class McpToolsCache(val servers: List<McpServer>)
    data class ActiveSelection(
        val selectedTools: List<String>,
        val instruction: String,
    )

    private fun cacheDir(project: Project): File {
        return File("${project.basePath}/.kilo/.cache")
    }

    fun readTools(project: Project): List<McpServer> {
        val file = File(cacheDir(project), "mcp-tools.json")
        if (!file.exists()) return emptyList()
        return try {
            val type = object : TypeToken<McpToolsCache>() {}.type
            val cache: McpToolsCache = gson.fromJson(file.readText(), type)
            cache.servers
        } catch (e: Exception) {
            emptyList()
        }
    }

    fun readActive(project: Project): ActiveSelection? {
        val file = File(cacheDir(project), "active-mcp.json")
        if (!file.exists()) return null
        return try {
            val type = object : TypeToken<ActiveSelection>() {}.type
            gson.fromJson(file.readText(), type)
        } catch (e: Exception) {
            null
        }
    }

    fun writeActive(project: Project, selectedTools: List<String>, instruction: String) {
        val dir = cacheDir(project)
        if (!dir.exists()) dir.mkdirs()
        val data = mapOf(
            "updatedAt" to java.util.Date().toString(),
            "selectedTools" to selectedTools,
            "instruction" to instruction,
        )
        File(dir, "active-mcp.json").writeText(gson.toJson(data), Charsets.UTF_8)
    }

    fun clearActive(project: Project) {
        val file = File(cacheDir(project), "active-mcp.json")
        file.delete()
    }
}
```

### 4.4 McpSelectorButton.kt

```kotlin
package com.dmc.prompt

import com.intellij.icons.AllIcons
import com.intellij.openapi.project.Project
import com.intellij.openapi.ui.popup.JBPopupFactory
import com.intellij.ui.awt.RelativePoint
import java.awt.event.ActionEvent
import javax.swing.JButton

class McpSelectorButton(private val project: Project) : JButton(AllIcons.Nodes.DataTables) {

    init {
        toolTipText = "选择 MCP 知识库"
        isFocusPainted = false
        isContentAreaFilled = false
        addActionListener { showPopup() }
    }

    private fun showPopup() {
        val servers = McpToolCache.readTools(project)
        if (servers.isEmpty()) {
            com.intellij.notification.Notification(
                "Kilo Code",
                "未检测到 MCP 工具。请先在 kilo.jsonc 中配置 MCP 服务器。",
                com.intellij.notification.NotificationType.WARNING
            ).notify(project)
            return
        }

        // 当前已选中的工具
        val active = McpToolCache.readActive(project)
        val selectedSet = active?.selectedTools?.toSet() ?: emptySet()

        // 构建选择面板
        val panel = McpSelectorPanel(servers, selectedSet) { selectedTools, instruction ->
            if (selectedTools.isEmpty()) {
                McpToolCache.clearActive(project)
            } else {
                McpToolCache.writeActive(project, selectedTools, instruction)
            }
        }

        val popup = JBPopupFactory.getInstance()
            .createComponentPopupBuilder(panel, null)
            .createPopup()

        popup.show(RelativePoint.getSouthWestOf(this))
    }
}
```

### 4.5 McpSelectorPanel.kt

```kotlin
package com.dmc.prompt

import com.intellij.icons.AllIcons
import com.intellij.ui.ColoredTreeCellRenderer
import com.intellij.ui.SimpleTextAttributes
import com.intellij.ui.components.JBScrollPane
import com.intellij.ui.treeStructure.Tree
import java.awt.BorderLayout
import java.awt.FlowLayout
import javax.swing.JButton
import javax.swing.JPanel
import javax.swing.JTextArea
import javax.swing.JTree
import javax.swing.tree.DefaultMutableTreeNode
import javax.swing.tree.DefaultTreeModel
import javax.swing.tree.TreeSelectionModel

class McpSelectorPanel(
    servers: List<McpToolCache.McpServer>,
    initiallySelected: Set<String>,
    private val onConfirm: (selectedTools: List<String>, instruction: String) -> Unit,
) : JPanel(BorderLayout()) {

    private val toolCheckBoxes = mutableMapOf<String, Boolean>()
    private val instructionArea = JTextArea(2, 30)

    init {
        // 构建树：服务器节点 → 工具节点
        val root = DefaultMutableTreeNode("MCP 服务器")
        for (server in servers) {
            val serverNode = DefaultMutableTreeNode(server)
            for (tool in server.tools) {
                toolCheckBoxes[tool.name] = initiallySelected.contains(tool.name)
                serverNode.add(DefaultMutableTreeNode(tool))
            }
            root.add(serverNode)
        }

        val tree = Tree(DefaultTreeModel(root))
        tree.selectionMode = TreeSelectionModel.SINGLE_TREE_SELECTION
        tree.cellRenderer = McpTreeCellRenderer(toolCheckBoxes)
        tree.isToggleClickEnabled = true

        // 点击切换勾选
        tree.addMouseListener(object : java.awt.event.MouseAdapter() {
            override fun mouseClicked(e: java.awt.event.MouseEvent) {
                val row = tree.getRowForLocation(e.x, e.y)
                if (row < 0) return
                val node = tree.getLastSelectedPathComponent() as? DefaultMutableTreeNode ?: return
                val tool = node.userObject as? McpToolCache.McpTool ?: return
                toolCheckBoxes[tool.name] = !(toolCheckBoxes[tool.name] ?: false)
                tree.repaint()
            }
        })

        val scrollPane = JBScrollPane(tree)
        scrollPane.preferredSize = java.awt.Dimension(400, 300)
        add(scrollPane, BorderLayout.CENTER)

        // 底部：指令输入 + 确认/取消
        val bottomPanel = JPanel(BorderLayout())
        instructionArea.text = "请优先使用以下工具检索相关信息后再回答"
        instructionArea.lineWrap = true
        instructionArea.wrapStyleWord = true
        bottomPanel.add(javax.swing.JLabel("引导指令:"), BorderLayout.NORTH)
        bottomPanel.add(instructionArea, BorderLayout.CENTER)

        val buttonRow = JPanel(FlowLayout(FlowLayout.RIGHT))
        val confirmButton = JButton("确定", AllIcons.Actions.MenuSaveall)
        confirmButton.addActionListener { confirm() }
        val cancelButton = JButton("取消", AllIcons.Actions.Cancel)
        cancelButton.addActionListener { findParentPopup()?.closeOk(null) }
        buttonRow.add(confirmButton)
        buttonRow.add(cancelButton)
        bottomPanel.add(buttonRow, BorderLayout.SOUTH)

        add(bottomPanel, BorderLayout.SOUTH)
    }

    private fun confirm() {
        val selected = toolCheckBoxes.filter { it.value }.keys.toList()
        onConfirm(selected, instructionArea.text.trim())
        findParentPopup()?.closeOk(null)
    }

    private fun findParentPopup(): com.intellij.openapi.ui.popup.JBPopup? {
        var parent = parent
        while (parent != null) {
            if (parent is com.intellij.openapi.ui.popup.JBPopup) return parent
            parent = parent.parent
        }
        return null
    }

    private class McpTreeCellRenderer(
        private val checkBoxes: Map<String, Boolean>,
    ) : ColoredTreeCellRenderer() {
        override fun customizeCellRenderer(
            tree: JTree,
            value: Any,
            selected: Boolean,
            expanded: Boolean,
            leaf: Boolean,
            row: Int,
            hasFocus: Boolean,
        ) {
            val node = value as? DefaultMutableTreeNode ?: return
            when (val obj = node.userObject) {
                is McpToolCache.McpServer -> {
                    icon = AllIcons.Nodes.Folder
                    append(obj.name, SimpleTextAttributes.REGULAR_ATTRIBUTES)
                    append("  (${obj.tools.size})", SimpleTextAttributes.GRAYED_ATTRIBUTES)
                }
                is McpToolCache.McpTool -> {
                    val isChecked = checkBoxes[obj.name] ?: false
                    icon = if (isChecked) AllIcons.Actions.Checked else AllIcons.Actions.Unchecked
                    append(obj.name, SimpleTextAttributes.REGULAR_ATTRIBUTES)
                    if (obj.description.isNotEmpty()) {
                        append("  - ${obj.description}", SimpleTextAttributes.GRAYED_SMALL_ATTRIBUTES)
                    }
                }
                is String -> {
                    append(obj, SimpleTextAttributes.REGULAR_ATTRIBUTES)
                }
            }
        }
    }
}

```

```
点击 MCP 按钮 → 取消所有勾选 → 确定 → 清除 active-mcp.json
→ AI 恢复默认行为（不特别引导 MCP 工具）
```

## 七、边界与异常

| 场景 | 处理 |
|---|---|
| mcp-tools.json 不存在 | 按钮弹窗提示"未检测到 MCP 工具" |
| mcp-tools.json 格式错误 | 静默返回空列表 |
| active-mcp.json 不存在 | 不注入引导指令（正常对话） |
| MCP 服务器断开 | tool.definition 不会包含该服务器工具 |
| 用户未选任何工具 | 清除 active-mcp.json，不注入 |
| 多个项目同时打开 | 各项目独立缓存（项目级 `.kilo/.cache/`） |
| 弹窗中指令为空 | 使用默认指令"请优先使用以下工具" |
